# Projeto 1 - Modelagem Estatística em Projeto de Social Media

# Abra o terminal ou prompt de comando, navegue até a pasta com os arquivos e execute o comando abaixo para criar um ambiente virtual:

conda create --name <nome_projeto> python=3.13

# Ative o ambiente:

conda activate <nome_projeto> (ou: source activate <nome_projeto>)

# Instale o pip e as dependências:

conda install pip
pip install -r requirements.txt
